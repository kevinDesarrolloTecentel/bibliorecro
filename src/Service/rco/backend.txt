Route::group([
    'prefix' => 'adm-user'
], function () {
    Route::post('/login', [AdminUsuarioV2Controller::class, 'login']);

    Route::group([
        'middleware' => ['auth:api']
    ], function () {
        Route::controller(AdminUsuarioV2Controller::class)->group(function () {
            // Gestión de usuarios
            Route::get('/listar', 'index');

            Route::post('/crear', 'store');
            Route::put('/actualizar/{id}', 'update');
            Route::delete('/eliminar/{id}', 'destroy');
            Route::post('/logout', 'logout');
            Route::get('/perfil', 'me');
            Route::post('/refrescar', 'refresh');
        });
    });
});

        Route::get('/mes', 'listarPrestamosMes');
        Route::get('/anio', 'listarPrestamosAño');
        Route::get('/{id}', 'show');

        // Creación
        Route::post('/create', 'store');
        Route::post('/nuevo', 'storeNew');

        // Actualización
        Route::put('/{id}', 'update');

        // Eliminación
        Route::delete('/{id}', 'destroy');
    });
